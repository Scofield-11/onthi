THI HẾT MÔN - CA 2, PHÒNG 301 C5 · THIB-0 (DẠNG 1.1 - HAI CON TRỎ ĐẾM CẶP CHÊNH LỆCH ≤ D)

Ghép cặp dây cáp (TH 1.1: Độ dài chênh lệch không quá D)

Giới hạn 0.5 giây mỗi test · bộ nhớ 256 MB · 10 test · tối đa 100.0 điểm.
Bài này 5 điểm trong tổng 10 điểm của đề. Điểm thang 100 của bài được quy tỉ lệ về số điểm này.

Kho vật tư viễn thông có N sợi dây cáp quang, sợi thứ i có chiều dài Lᵢ mi-li-mét.

Hai sợi dây cáp ghép được thành một cặp cáp đôi nếu độ dài của chúng chênh nhau không quá D mi-li-mét (tức là |Lᵢ - Lⱼ| ≤ D) — khi đó kỹ thuật viên có thể cân chỉnh đầu bấm coi như bằng nhau.

Hãy đếm xem có bao nhiêu cách chọn ra một cặp gồm 2 sợi cáp có thể ghép được với nhau. Hai sợi cáp ở hai vị trí khác nhau thì tính là hai sợi khác nhau, kể cả khi chúng có cùng độ dài (tức là đếm số cặp chỉ số (i, j) thỏa mãn 1 ≤ i < j ≤ N).

Dữ liệu vào

  • Dòng 1: N và D (1 ≤ N ≤ 200 000; 0 ≤ D ≤ 10⁹).
  • Dòng 2: N số nguyên Lᵢ (1 ≤ Lᵢ ≤ 10⁹).

Kết quả ra

Một số nguyên — số cặp dây cáp ghép được. Kết quả có thể tới khoảng 2·10¹⁰, vượt int, nên bắt buộc dùng long long.

Ví dụ

5 3 10 4 7 1 9

Kết quả: 5 — các cặp độ dài ghép được là (1,4), (4,7), (7,9), (7,10), (9,10).

💡 Gợi ý thuật toán & Bẫy cần tránh:
• Subtask 1 (N ≤ 1 000 - 3 điểm): Dùng 2 vòng lặp for lồng nhau (i từ 0 đến N-1, j từ i+1 đến N-1), kiểm tra abs(L[i] - L[j]) <= D.
• Subtask 2 (N ≤ 200 000 - đủ 5 điểm): Sắp xếp mảng L tăng dần bằng sort(L, L + n). Dùng kỹ thuật 2 con trỏ: với mỗi đầu phải r từ 0 đến n-1, dịch đầu trái l tăng lên chừng nào L[r] - L[l] > D. Khi đó mọi chỉ số từ l đến r-1 đều ghép được với r, cộng (r - l) vào biến đếm long long ans. Độ phức tạp O(N log N). 
#include <bits/stdc++.h>
using namespace std;

int main() {
    // Tăng tốc độ nhập xuất chuẩn C++
    ios_base::sync_with_stdio(false);
    cin.tie(NULL);

    int n;
    long long d;
    if (!(cin >> n >> d)) return 0;

    vector<long long> L(n);
    for (int i = 0; i < n; i++) {
        cin >> L[i];
    }

    // Bước 1: Sắp xếp độ dài các sợi cáp tăng dần -> O(N log N)
    sort(L.begin(), L.end());

    // Bước 2: Kỹ thuật 2 con trỏ (Two Pointers) -> O(N)
    long long dem = 0; // BẮT BUỘC dùng long long vì tối đa có N*(N-1)/2 ~ 2*10^10 cặp
    int l = 0;
    for (int r = 0; r < n; r++) {
        while (L[r] - L[l] > d) {
            l++;
        }
        // Các sợi từ vị trí l đến r-1 đều có L[r] - L[i] <= d
        dem += (r - l);
    }

    cout << dem << "\n";
    return 0;
}

• Bẫy K = 0: Nếu dùng tìm kiếm nhị phân trên toàn mảng mà không giới hạn vùng tìm kiếm từ i + 1 trở đi, khi K = 0 chương trình sẽ đếm cả chính nó (i == j).

#include <bits/stdc++.h>
using namespace std;

int main() {
    ios_base::sync_with_stdio(false);
    cin.tie(NULL);

    int n;
    long long k;
    if (!(cin >> n >> k)) return 0;

    vector<long long> L(n);
    for (int i = 0; i < n; i++) {
        cin >> L[i];
    }

    // Sắp xếp tăng dần để với mọi j > i ta luôn có L[j] >= L[i]
    sort(L.begin(), L.end());

    long long dem = 0;
    for (int i = 0; i < n; i++) {
        long long can_tim = L[i] + k;
        // Chỉ tìm ở các vị trí j > i (tức từ L.begin() + i + 1 đến L.end())
        // Cách này xử lý đúng hoàn toàn cả trường hợp K = 0
        auto it1 = lower_bound(L.begin() + i + 1, L.end(), can_tim);
        auto it2 = upper_bound(L.begin() + i + 1, L.end(), can_tim);
        dem += (it2 - it1);
    }

    cout << dem << "\n";
    return 0;
}

#include <bits/stdc++.h>
using namespace std;

long long p[200005], temp[200005];

// Hàm Merge Sort vừa sắp xếp tăng dần vừa trả về số cặp nghịch thế trong đoạn [l, r]
long long demNghichThe(int l, int r) {
    if (l >= r) return 0;
    int mid = (l + r) / 2;
    long long dem = demNghichThe(l, mid) + demNghichThe(mid + 1, r);

    int i = l, j = mid + 1, k = l;
    while (i <= mid && j <= r) {
        if (p[i] <= p[j]) {
            temp[k++] = p[i++];
        } else {
            // p[i] > p[j] => Mọi phần tử từ p[i] đến p[mid] đều > p[j]
            dem += (mid - i + 1);
            temp[k++] = p[j++];
        }
    }
    while (i <= mid) temp[k++] = p[i++];
    while (j <= r) temp[k++] = p[j++];
    for (int idx = l; idx <= r; idx++) p[idx] = temp[idx];

    return dem;
}

// Có N cuộn cáp dài L[i], cần cắt ra ít nhất K đoạn cáp dài bằng nhau.
// Tìm độ dài nguyên lớn nhất của mỗi đoạn cắt được.
bool kiemTra(const vector<long long>& L, long long k, long long do_dai) {
    long long so_doan = 0;
    for (long long x : L) {
        so_doan += (x / do_dai);
        if (so_doan >= k) return true;
    }
    return false;
}

long long mergeCount(int l, int r) {
    if (l >= r) return 0;
    int mid = (l + r) / 2;
    long long inv_count = mergeCount(l, mid) + mergeCount(mid + 1, r);

    int i = l, j = mid + 1, k = l;
    while (i <= mid && j <= r) {
        if (a[i] <= a[j]) {
            temp[k++] = a[i++];

    while (j <= r) temp[k++] = a[j++];
    for (int idx = l; idx <= r; idx++) a[idx] = temp[idx];

    return inv_count;
}

int main() {

    cin.tie(NULL);

    if (!(cin >> n)) return 0;

    vector<int> a(n);
    for (int i = 0; i < n; i++) {
        cin >> a[i];
    }

    stack<int> st;
    int idx = 0; // Chỉ số toa cần xuất ra ở mảng a

    for (int toa_vao = 1; toa_vao <= n; toa_vao++) {
        st.push(toa_vao);
        // Chừng nào toa ở đỉnh đường ray cụt khớp với toa cần xuất tiếp theo thì đẩy ra ngay
        while (!st.empty() && st.top() == a[idx]) {
            st.pop();
            idx++;
        }
    }

    if (st.empty()) {
        cout << "YES\n";
    } else {
        cout << "NO\n";
    }
    return 0;
}

    stack<int> st;
    int can_ra = 1;

    for (int i = 0; i < n; i++) {
        st.push(a[i]);
        while (!st.empty() && st.top() == can_ra) {
            st.pop();
            can_ra++;
        }
    }

    cout << (st.empty() ? "YES\n" : "NO\n");
    return 0;
}
